//Mediator Pattern (MediatR)

//Mediator pattern is used to reduce tight coupling between controller and business logic.
//Instead of directly injecting Commands/Queries into controller,
//controller will communicate only with IMediator.

//Before mediator

Controller
   -> Query / Command
        -> Service
            -> Repository

//After mediator

Controller
   -> IMediator
        -> Handler
             -> Service
                  -> Repository

//Main advantage:
//controller becomes very thin and clean

//Before

private readonly GetAllExpenseQuery _getAllExpenseQuery;
private readonly CreateExpenseCommand _createExpenseCommand;
private readonly UpdateExpenseCommand _updateExpenseCommand;

//After

private readonly IMediator _mediator;

//Mediator automatically finds correct handler and executes it.

//IRequest<T>

//IRequest<T> represents a request sent through mediator
//T represents the response type

//Syntax

IRequest<T>

//Example

public class GetAllExpenseQuery : IRequest<List<ExpenseDto>>
{
    public string? SearchTerm { get; set; }
}

//Meaning:
//GetAllExpenseQuery will return List<ExpenseDto>

//Examples

IRequest<int>
//returns integer

IRequest<UserDto>
//returns UserDto

IRequest<List<ExpenseDto>>
//returns list of expenses

//------------------------------------------------------------

//IRequestHandler<TRequest, TResponse>

//This handles/processes the request

//Syntax

IRequestHandler<RequestType, ReturnType>

//Example

public class GetAllExpenseQueryHandler
    : IRequestHandler<GetAllExpenseQuery, List<ExpenseDto>>
{
}

//Meaning:
//This handler processes GetAllExpenseQuery
//and returns List<ExpenseDto>

//Mediator internally matches request with correct handler

//Example

GetAllExpenseQuery
      ->
GetAllExpenseQueryHandler

//------------------------------------------------------------

//Handle() method

//Handle method contains actual business logic
//Mediator automatically calls this method internally

//Syntax

public async Task<TResponse> Handle(
    TRequest request,
    CancellationToken cancellationToken
)
{
}

//Example

public async Task<List<ExpenseDto>> Handle(
    GetAllExpenseQuery request,
    CancellationToken cancellationToken
)
{
    var query = _expenseService.GetAllExpenses();

    return query.ToList();
}

//When mediator receives GetAllExpenseQuery,

//CancellationToken

//CancellationToken is used to stop/cancel an operation
//when the request is no longer needed.

//Why needed?

//Sometimes:
//1. User closes browser
//2. Request timeout happens
//3. Client disconnects
//4. Long DB query still running

//Without CancellationToken:
//DB/API operation still continues
//Server resources get wasted

//With CancellationToken:
//ASP.NET Core can stop operation gracefully

//------------------------------------------------------------

//Main Benefits

//1. Prevents unnecessary work
//2. Saves server resources
//3. Improves scalability
//4. Industry standard in async APIs
//5. Useful for DB/API/file operations

//------------------------------------------------------------

//Request flow

Client
   ->
Controller
   ->
Mediator
   ->
Handler
   ->
Service
   ->
EF Core / Database

HttpContext.RequestAborted

public async Task<List<ExpenseDto>> GetAllExpensesAsync(

)
{

        .AsNoTracking()
        .ToListAsync(cancellationToken);
}

//Parquet File Export

Visual representation

Parquet (column-based)
Id       -> [1,2,3]

Category -> [Food,Travel,Food]

Amount   -> [500,800,200]

//Parquet is a column-based binary file format
//mainly used in analytics and big-data systems.

//Unlike CSV:
//CSV stores row by row

//Example:
//1,Food,500
//2,Travel,700

//Parquet stores column by column

//Id -> 1,2
//Category -> Food,Travel
//Amount -> 500,700

//Because of this:
//1. Better compression
//2. Faster analytical queries
//3. Less storage usage
//4. Schema support
//5. Widely used in Spark/Hadoop/Data Lakes

using Expense.Application.Contracts;

using Parquet.Serialization;

namespace Expense.Infrastructure.Exporters
{
public class ParquetExpenseExporter : IExpenseExporter
{
public string ContentType => "application/vnd.apache.parquet";

```
    public string FileExtension => "parquet";

    public byte[] Export(IEnumerable<ExpenseDto> expenses)
    {
        using var stream = new MemoryStream();

        ParquetSerializer
            .SerializeAsync(expenses, stream)
            .GetAwaiter()
            .GetResult();

        return stream.ToArray();
    }
}
```

}
