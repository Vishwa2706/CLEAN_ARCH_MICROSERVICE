dotnet new webapi -n Expense.API

//Add references 

dotnet add Expense.API reference Expense.Application
dotnet add Expense.API reference Expense.Infrastructure

dotnet add Expense.Application reference Expense.Domain

dotnet add Expense.Infrastructure reference Expense.Application


//Add ef core package, if error thorws add with specific package

dotnet add Expense.Infrastructure package Microsoft.EntityFrameworkCore
dotnet add Expense.Infrastructure package Microsoft.EntityFrameworkCore.SqlServer

//Add connection string in appsettings.json

//TO run migrations at first to create a new table in DB, add context in Repository and create model/entities in domain(dto)

//Add tbcontext injection in API layer in program.classlib

using User.Infrastructure.Repository;
using Microsoft.EntityFrameworkCore;

var builder = WebApplication.CreateBuilder(args);

// Add DbContext
builder.Services.AddDbContext<ExpenseRepository>(options =>
    options.UseNpgsql(builder.Configuration.GetConnectionString("DefaultConnection"))
);

var app = builder.Build();

app.UseSwagger();
app.UseSwaggerUI();
app.MapControllers();

app.Run();

//run below commands to apply migrations

dotnet ef migrations add AddUserTable --project Expense.Infrastructure --startup-project Expense.API
dotnet ef database update --project Expense.Infrastructure --startup-project Expense.API


steps to write a simple API,(personal use)

1.create dto/model/entities in domain if needed
2.create interface and add your method there
3.create service and implement this interface and query using LINQ
4.go to application layer and based on CQRS pattern create a get.cs file and inject Interface there and call method inside that service
5.map this inteface and infra sevrice in program.cs
6.go to controller and inject query file from application and call the method

//normal
namespace User.Domain.Models
{
    public class UserDto
    {
        public int Id { get; set; }

        public string Name { get; set; } = null!;

        public int Mobile { get; set; }

        public ICollection<Expense> Expenses { get; set; } = new List<Expense>();
    }
}

//reverse 
namespace Expense.Domain.Models
{
    public class Expense
    {
        public int Id { get; set; }

        // Foreign key
        public int UserId { get; set; }

        public string Category { get; set; } = string.Empty;
        public decimal Amount { get; set; }
        public DateTime Date { get; set; }
        public string? Note { get; set; }

        // Navigation property
        public User User { get; set; } = null!;
    }
}
