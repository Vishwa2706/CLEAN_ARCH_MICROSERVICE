    {
        RuleFor(x => x.Request.Category)

        RuleFor(x => x.Request.Amount)

            .WithMessage("Amount must be greater than zero");

        RuleFor(x => x.Request.UserId)

            .WithMessage("User id must be greater than zero");
    }

//ValidationBehavior

//ValidationBehavior automatically executes validators
//before handler execution

using FluentValidation;
using MediatR;
using Shared.Exceptions;

namespace Expense.Application.Validator;

public class ValidationBehavior<TRequest, TResponse>
    : IPipelineBehavior<TRequest, TResponse>
    where TRequest : IRequest<TResponse>
{
    private readonly IEnumerable<IValidator<TRequest>> _validators;

    public ValidationBehavior(
        IEnumerable<IValidator<TRequest>> validators
    )
    {
        _validators = validators;
    }

    public async Task<TResponse> Handle(
        TRequest request,
        RequestHandlerDelegate<TResponse> next,
        CancellationToken cancellationToken
    )
    {
        var context = new ValidationContext<TRequest>(request);

        var validationTasks = _validators
            .Select(v => v.ValidateAsync(context, cancellationToken))
            .ToArray();

        var validationResults = await Task.WhenAll(validationTasks);

        var failures = validationResults
            .SelectMany(result => result.Errors)
            .Where(f => f != null)
            .ToList();

        if (failures.Count != 0)
        {
            var errorMessages = string.Join(
                "; ",
                failures.Select(f => f.ErrorMessage)
            );

            throw new BadRequestException(
                "Validation Failed",
                errorMessages,
                "VALIDATION_ERROR"
            );
        }

        return await next();
    }
}

1.Why add BeginTransactionAsync to IExpenseService?
To expose a PostgreSQL transaction so Save Expense and Publish RabbitMQ can be coordinated as one business operation.

2.What happens in the success path?
Begin transaction → save expense → publish RabbitMQ event → commit transaction.

3.What happens if RabbitMQ publish throws an exception?
The catch block rolls back the database transaction, so the expense is not persisted.

4.Does this guarantee atomicity across DB and RabbitMQ?
No. PostgreSQL and RabbitMQ are separate systems, so this reduces inconsistencies but does not fully eliminate distributed transaction edge cases.

5.What are the main edge cases that still exist?
Message published but producer never receives confirmation, and app crash after DB commit but before publish.

6.Why is the Outbox Pattern preferred in production?
Because it guarantees that if business data is committed, the corresponding event is also persisted and can be published later.

//Interview Answer

What is Idempotency?

Idempotency ensures that processing
the same message multiple times
produces the same result.

RabbitMQ provides At-Least-Once
Delivery, so duplicate messages
can occur.

To prevent duplicate processing,
we generate a unique EventId,
store processed EventIds in a table,
and ignore messages that have
already been processed.
